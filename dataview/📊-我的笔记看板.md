function loadConfig() {
    const fm = dv.current()?.file?.frontmatter;
    const cfg = { ...DEFAULT_CONFIG };
    const savedScheme = localStorage.getItem('nd-color-scheme');
    if (savedScheme && COLOR_SCHEMES[savedScheme]) cfg.colorScheme = savedScheme;
    const cachedCfg = localStorage.getItem('nd-config-cache');
    if (cachedCfg) {
        try {
            const cc = JSON.parse(cachedCfg);
            for (const [k, v] of Object.entries(cc)) {
                if (k !== 'colorScheme' && k in DEFAULT_CONFIG && v !== undefined && v !== null) cfg[k] = v;
            }
        } catch {}
    }
    if (!fm) return cfg;
    if (Array.isArray(fm.exclude) && fm.exclude.length > 0) cfg.exclude = fm.exclude;
    if (typeof fm.planPath === 'string' && fm.planPath) cfg.planPath = fm.planPath;
    if (Array.isArray(fm.heatLevels) && fm.heatLevels.length === 3 && fm.heatLevels.every(n => typeof n === 'number'))
        cfg.heatLevels = [...fm.heatLevels].sort((a, b) => a - b);
    if (typeof fm.heatWeeks === 'number' && fm.heatWeeks >= 10 && fm.heatWeeks <= 104) cfg.heatWeeks = Math.round(fm.heatWeeks);
    if (typeof fm.monthCount === 'number' && fm.monthCount >= 1 && fm.monthCount <= 24) cfg.monthCount = Math.round(fm.monthCount);

    if (typeof fm.folderTopN === 'number' && fm.folderTopN >= 1 && fm.folderTopN <= 50) cfg.folderTopN = Math.round(fm.folderTopN);
    if (typeof fm.maxOpen === 'number' && fm.maxOpen >= 0 && fm.maxOpen <= 20) cfg.maxOpen = Math.round(fm.maxOpen);

function forcePreviewMode() {
    dv.app.workspace.iterateAllLeaves(l => {
        if (!l?.view?.file) return;
        if (l.view.file.path !== C.selfPath) return;
        // 跳过还没有完整 viewState 的 leaf
        const st = l.getViewState();
        if (!st || !st.state || typeof st.state.mode === 'undefined') return;
        if (st.state.mode !== 'preview') {
            l.setViewState({ ...st, state: { ...(st.state || {}), mode: 'preview' } });
        }
    });
}

// 非 preview 模式时清空容器并切换
if (!ndIsPre) {
    dv.container.innerHTML = '';
    let retries = 0;
    const MAX_RETRIES = 5;
    const switchToPreview = () => {
        forcePreviewMode();
        // 检查是否真的生效了（通过比对当前 mode），未生效则重试
        let switched = false;
        dv.app.workspace.iterateAllLeaves(l => {
            if (l?.view?.file?.path !== C.selfPath) return;
            if (l.getViewState()?.state?.mode === 'preview') switched = true;
        });
        if (!switched && retries < MAX_RETRIES) {
            retries++;
            setTimeout(switchToPreview, 50 * retries);  // 退避重试 50ms, 100ms, 150ms...
        }
    };
    // 用 dv.app.workspace.onLayoutReady 确保 workspace 可安全操作
    dv.app.workspace.onLayoutReady(() => {
        requestAnimationFrame(() => {
            switchToPreview();
        });
    });
    return;
}

                    row.style.transition = 'opacity .25s ease, transform .25s ease';
                    row.style.opacity = '1';
                    row.style.transform = '';
                }, 40 + i * childDelay);
            });
        }
        el.addEventListener('transitionend', cleanup, { once: true });
        setTimeout(cleanup, 500); // fallback
    } else {
        const h = el.scrollHeight;
        el.style.maxHeight = h + 'px';
        el.style.transition = 'none';
        if (padding) { el.style.paddingTop = padding ? '4px' : ''; el.style.paddingBottom = padding ? '6px' : ''; }
        void el.offsetHeight;
        el.style.transition = 'max-height .3s cubic-bezier(.4,0,.2,1), opacity .2s ease .1s'
            + (padding ? ', padding .25s ease .05s' : '');
        el.style.maxHeight = '0px';
        el.style.opacity = '0';
        if (padding) { el.style.paddingTop = '0px'; el.style.paddingBottom = '0px'; }
        const onEnd = () => {
            el.classList.remove('nd-show');
            cleanup();
        };
        el.addEventListener('transitionend', onEnd, { once: true });
        setTimeout(onEnd, 500); // fallback
    }
}

function bindChipInputs(panel) {
    panel.querySelectorAll('[data-nd-chip]').forEach(container => {
        const name = container.dataset.ndChip;
        const input = container.querySelector(`[data-nd-chip-input="${name}"]`);
        input.addEventListener('keydown', e => {
            if (e.key !== 'Enter' && e.key !== ',') return;
            e.preventDefault();
            const val = input.value.trim().replace(/,$/, '');
            if (!val) return;
            const chips = getChipValues(panel, name);
            if (chips.includes(val)) { input.value = ''; return; }
            chips.push(val);
            renderChips(container, name, chips);
            input.value = '';
        });
        container.addEventListener('click', e => {
            const del = e.target.closest('[data-nd-chip-del]');
            if (del) {
                const idx = parseInt(del.dataset.ndChipDel);
                const chips = getChipValues(panel, name);
                chips.splice(idx, 1);
                renderChips(container, name, chips);
                return;
            }
            input.focus();
        });
    });
}

function bindSectionOrderButtons(panel) {
    const container = panel.querySelector('[data-nd-section-order]');
    if (!container) return;

    container.addEventListener('dragstart', e => { e.preventDefault(); e.stopPropagation(); });

    const springEase = 'cubic-bezier(.22,1.2,.36,1)';
    const SPRING_MS = 420;

    const getItems = () => [...container.querySelectorAll('[data-nd-section]')];

    async function moveAndAnimate(insertFn) {
        const allEls = getItems();
        allEls.forEach(el => { el.style.transition = 'none'; el.style.transform = ''; });
        void container.offsetWidth;

        const oldRects = new Map();
        allEls.forEach(el => oldRects.set(el, el.getBoundingClientRect()));

        insertFn();
        void container.offsetWidth;

        const newEls = getItems();
        const toAnimate = [];
        newEls.forEach(el => {
            const oldR = oldRects.get(el);
            if (!oldR) return;
            const newR = el.getBoundingClientRect();
            const dy = oldR.top - newR.top;
            if (Math.abs(dy) > 0.5) toAnimate.push(el);
        });

        toAnimate.forEach((el, i) => {
            const oldR = oldRects.get(el);
            const newR = el.getBoundingClientRect();
            const dy = oldR.top - newR.top;
            el.style.transition = 'none';
            el.style.transform = `translateY(${dy}px)`;
        });
        void container.offsetWidth;
        toAnimate.forEach((el, i) => {
            el.style.transition = `transform ${SPRING_MS}ms ${springEase} ${i * 18}ms`;
            el.style.transform = '';
        });

        setTimeout(() => {
            newEls.forEach(el => { el.style.transition = ''; el.style.transform = ''; });
        }, SPRING_MS + toAnimate.length * 18 + 60);
    }

    let isDragging = false;
    let currentItem = null;
    let clone = null;
    let offsetY = 0;
    let animFrame = null;

    function onPointerDown(e) {
        const handle = e.target.closest('.nd-drag-handle');
        if (!handle) return;
        const item = handle.closest('[data-nd-section]');
        if (!item) return;
        e.preventDefault();
        e.stopPropagation();

        currentItem = item;
        isDragging = true;

        const clientY = e.touches ? e.touches[0].clientY : e.clientY;
        const rect = item.getBoundingClientRect();
        offsetY = clientY - rect.top;

        clone = item.cloneNode(true);
        clone.style.cssText = `position:fixed;left:${rect.left}px;top:${rect.top}px;width:${rect.width}px;height:${rect.height}px;opacity:.92;transform:scale(1.05);box-shadow:0 14px 36px rgba(0,0,0,.22),0 0 0 2px var(--interactive-accent);border-radius:8px;z-index:9999;pointer-events:none;transition:transform .25s ${springEase},box-shadow .25s;background:var(--background-primary);`;
        document.body.appendChild(clone);

        item.style.transition = 'opacity .2s,transform .2s';
        item.style.opacity = '0.2';
        item.style.transform = 'scale(0.95)';
        getItems().forEach(el => { if (el !== item) { el.style.transition = 'opacity .2s'; el.style.opacity = '0.8'; } });

        document.addEventListener('mousemove', onPointerMove);
        document.addEventListener('mouseup', onPointerUp);
        document.addEventListener('touchmove', onPointerMove, { passive: false });
        document.addEventListener('touchend', onPointerUp);
    }

    function onPointerMove(e) {
        if (!isDragging || !currentItem || !clone) return;
        e.preventDefault();
        const clientY = e.touches ? e.touches[0].clientY : e.clientY;

        if (animFrame) cancelAnimationFrame(animFrame);
        animFrame = requestAnimationFrame(() => {
            if (!clone) return;
            clone.style.top = (clientY - offsetY) + 'px';

            const after = getDragAfterElement(container, clientY);
            let moved = false;
            if (after === null) {
                if (currentItem !== container.lastElementChild) moved = true;
            } else if (after !== currentItem) {
                moved = true;
            }
            if (moved) {
                moveAndAnimate(() => {
                    if (after === null) container.appendChild(currentItem);
                    else container.insertBefore(currentItem, after);
                });
            }
        });
    }

    function onPointerUp() {
        if (!isDragging) return;
        isDragging = false;
        if (animFrame) { cancelAnimationFrame(animFrame); animFrame = null; }

        const item = currentItem;

        if (clone && item) {
            const rect = item.getBoundingClientRect();
            clone.style.transition = `all .3s ${springEase}`;
            clone.style.top = rect.top + 'px';
            clone.style.transform = 'scale(1)';
            clone.style.opacity = '0';
            const c = clone; setTimeout(() => c.remove(), 320);
            clone = null;
        }

        if (item) {
            item.style.transition = 'opacity .25s,transform .25s';
            item.style.opacity = '';
            item.style.transform = '';
            getItems().forEach(el => {
                if (el !== item) { el.style.transition = 'opacity .25s'; el.style.opacity = ''; }
                el.style.transform = '';
            });
            setTimeout(() => getItems().forEach(el => { el.style.transition = ''; }), 260);
            currentItem = null;
        }

        document.removeEventListener('mousemove', onPointerMove);
        document.removeEventListener('mouseup', onPointerUp);
        document.removeEventListener('touchmove', onPointerMove);
        document.removeEventListener('touchend', onPointerUp);
    }

    container.addEventListener('mousedown', onPointerDown);
    container.addEventListener('touchstart', onPointerDown, { passive: false });

    function getDragAfterElement(cont, y) {
        const items = getItems().filter(el => el !== currentItem);
        let closest = null, closestDist = Infinity;
        for (const child of items) {
            const box = child.getBoundingClientRect();
            const center = box.top + box.height / 2;
            const dist = y - center;
            if (dist < 0 && -dist < closestDist) { closestDist = -dist; closest = child; }
        }
        return closest;
    }

    container.querySelectorAll('.nd-drag-handle').forEach(handle => {
        handle.addEventListener('mouseenter', () => { handle.style.color = 'var(--interactive-accent)'; handle.style.background = 'var(--background-primary-alt)'; handle.style.transform = 'scale(1.1)'; });
        handle.addEventListener('mouseleave', () => { handle.style.color = ''; handle.style.background = ''; handle.style.transform = ''; });
    });
}

function renderChips(container, name, values) {
    const input = container.querySelector(`[data-nd-chip-input="${name}"]`);
    container.querySelectorAll('[data-nd-chip-val]').forEach(el => el.remove());
    values.forEach((v, i) => {

}

function collectFormValues(panel) {
    const newCfg = { ...DEFAULT_CONFIG };
    panel.querySelectorAll('[data-nd-field]').forEach(input => {
        const name = input.dataset.ndField;
        if (input.type === 'number') {
            const val = parseFloat(input.value);
            if (!isNaN(val)) {
                if (name.includes('[')) {
                    const [arr, idx] = name.replace(']', '').split('[');
                    if (!Array.isArray(newCfg[arr])) newCfg[arr] = [...DEFAULT_CONFIG[arr]];
                    newCfg[arr][parseInt(idx)] = val;
                } else {
                    newCfg[name] = val;
                }
            }
        } else {
            newCfg[name] = input.value.trim();
        }
    });
    newCfg.exclude = getChipValues(panel, 'exclude');
    if (newCfg.exclude.length === 0) newCfg.exclude = [...DEFAULT_CONFIG.exclude];
    newCfg.taskTags = getChipValues(panel, 'taskTags');
    newCfg.sectionOrder = getSectionOrder(panel);
    const checked = panel.querySelector('input[name="nd-color"]:checked');
    if (checked) newCfg.colorScheme = checked.value;
    newCfg.heatLevels.sort((a, b) => a - b);
    return newCfg;
}

const WordCache = {
    CACHE_KEY: 'nd-word-cache-v1',
    _cache: null,
    _fileKey(p) { return p.file.path + ':' + p.file.size + ':' + (p.file.mtime || 0); },
    _wordCount(raw) {
        if (!raw) return 0;
        // 正则优化：[^\]\n]* 禁止跨行匹配减少回溯；合并行首标记
        let t = raw.replace(/^---[\s\S]*?---\n*/, '')
            .replace(/```[\s\S]*?```/g, '')
            .replace(/`[^`\n]+`/g, '')
            .replace(/!\[\[[^\]\n]*\]\]/g, '')
            .replace(/\[\[([^\]|#\n^]+)(?:[#|][^\]]*)?\]\]/g, '$1')
            .replace(/\[([^\]\n]*)\]\([^)\n]+\)/g, '$1')
            .replace(/([\*_]{1,3})([^\*_]*?)\1/g, '$2')
            .replace(/^#+\s*/gm, '')
            .replace(/^[\s>*+-]\s*/gm, '')
            .replace(/^\d+\.\s*/gm, '');
        const cjk = (t.match(/[\u4e00-\u9fff\u3400-\u4dbf\uf900-\ufaff]/g) || []).length;
        const other = t.replace(/[\u4e00-\u9fff\u3400-\u4dbf\uf900-\ufaff]/g, ' ').split(/\s+/).filter(w => w).length;
        return cjk + other;
    },

    async load(pages) {
        let cached = {};
        try { cached = JSON.parse(localStorage.getItem(this.CACHE_KEY) || '{}'); } catch {}

        const wordByPath = new Map();
        let totalWords = 0;
        const toLoad = [];

        for (const p of pages.values) {
            const fk = this._fileKey(p);
            if (cached[fk] !== undefined) {
                wordByPath.set(p.file.path, cached[fk]);
                totalWords += cached[fk];
            } else {
                toLoad.push(p);
            }
        }

        // 大笔记库走估算
        if (toLoad.length > cfg.estThreshold) {
            for (const p of toLoad) {
                const est = Math.round((p.file.size || 0) / cfg.estCoeff);
                wordByPath.set(p.file.path, est);
                totalWords += est;
                cached[this._fileKey(p)] = est;
            }
            localStorage.setItem(this.CACHE_KEY, JSON.stringify(cached));
            this._cache = { wordByPath, totalWords };
            return this._cache;
        }

        // 只加载变化的文件
        if (toLoad.length > 0) {
            const raws = await Promise.all(toLoad.map(async p => {
                try { return await dv.io.load(p.file.path); } catch { return ''; }
            }));
            for (let i = 0; i < toLoad.length; i++) {
                const wc = this._wordCount(raws[i] || '');
                wordByPath.set(toLoad[i].file.path, wc);
                totalWords += wc;
                cached[this._fileKey(toLoad[i])] = wc;
            }
            // 清理旧缓存（保留最近500条）
            const keys = Object.keys(cached);
            if (keys.length > 500) {
                keys.sort();
                for (let i = 0; i < keys.length - 500; i++) delete cached[keys[i]];
            }
            localStorage.setItem(this.CACHE_KEY, JSON.stringify(cached));
        }

        this._cache = { wordByPath, totalWords };
        return this._cache;
    }
};

// 月度统计 / 最近7天统计（可切换纵向柱状图）
function niceScale(maxVal, ticks) {
    if (maxVal <= 0) return { max: 100, step: 25, ticks: [0, 25, 50, 75, 100] };
    const rawStep = maxVal / ticks;
    const pow10 = Math.pow(10, Math.floor(Math.log10(rawStep)));
    const candidates = [pow10, pow10 * 2, pow10 * 5, pow10 * 10];
    let niceStep = candidates[0];
    for (const c of candidates) { if (c * ticks >= maxVal) { niceStep = c; break; } }
    const tickValues = [];
    for (let i = 0; i <= ticks; i++) tickValues.push(Math.round(i * niceStep * 1000) / 1000);
    return { max: niceStep * ticks, step: niceStep, ticks: tickValues };
}

function bindToggleWithAnim(toggleId, viewAKey, viewBKey, paneAId, paneBId, titleAText, titleBText, onSwitch) {
    requestAnimationFrame(() => {
        const toggleEl = document.getElementById(toggleId);
        if (!toggleEl) return;
        const paneA = document.getElementById(paneAId);
        const paneB = document.getElementById(paneBId);
        const titleEl = document.getElementById(toggleId + '-title');
        setupToggle(toggleEl, (view) => {
            const isB = view === viewBKey;
            const showPane = isB ? paneB : paneA;
            const hidePane = isB ? paneA : paneB;
            const newTitle = isB ? titleBText : titleAText;
            if (onSwitch) onSwitch(view, showPane, hidePane); else showPaneWithAnim(showPane, hidePane);
            if (titleEl) {
                titleEl.style.transition = 'opacity .25s ease';
                titleEl.style.opacity = '0';
                setTimeout(() => { titleEl.textContent = newTitle; titleEl.style.opacity = '1'; }, 150);
            }
        });
    });
}

// 通用toggle切换函数
function setupToggle(toggleEl, onSwitch) {
    const indicator = toggleEl.querySelector('[data-toggle-indicator]');
    const spans = toggleEl.querySelectorAll('span[data-toggle-view]');
    let currentView = null;
    spans.forEach(sp => {
        sp.addEventListener('click', function(e) {
            e.stopPropagation();
            const view = this.dataset.toggleView;
            if (view === currentView) return;
            currentView = view;
            if (indicator) indicator.style.left = view === spans[spans.length-1].dataset.toggleView ? 'calc(50%)' : '2px';
            spans.forEach(s => { s.style.color = s === this ? '#fff' : 'var(--text-muted)'; });
            onSwitch(view, this);
        });
    });
    // 初始化当前视图
    currentView = spans[0]?.dataset.toggleView;
}

function getTimeDiff(date) {
    const now = dv.date('now');
    const diff = now.ts - date.ts;
    if (diff < 0) return '刚刚'; // 未来日期（时钟偏差）视为刚刚
    const mins = Math.floor(diff / 60000);
    if (mins < 1) return '刚刚';
    if (mins < 60) return `${mins}分钟前`;
    const hours = Math.floor(mins / 60);
    if (hours < 24) return `${hours}小时前`;
    const days = Math.floor(hours / 24);
    if (days < 7) return `${days}天前`;
    return `${date.month}/${date.day}`;
}

// 辅助函数
const _tagPattern = new RegExp(cfg.taskTags.map(t => t.replace(/[.*+?^${}()|[\]\\]/g, '\\$&') + '(?!\\w)').join('|'), 'gi');
const _analyzeCache = new Map();
function analyzeTask(task) {
    const key = task.path + ':' + task.line;
    if (_analyzeCache.has(key)) return _analyzeCache.get(key);
    const text = task.text || '';
    const isUrgent = /!!/.test(text);
    const isImportant = (/!(?!!)/.test(text) && !/!!/.test(text)) || /#important/i.test(text);
    const dueMatch = text.match(/📅\s*(\d{4}-\d{2}-\d{2})/);
    const dueDate = dueMatch ? new Date(dueMatch[1]) : null;
    let dueStatus = 'none';
    if (dueDate) {
        const diff = Math.ceil((dueDate - C.now) / (1000*60*60*24));
        if (diff < 0) dueStatus = 'overdue';
        else if (diff === 0) dueStatus = 'today';
        else if (diff <= 2) dueStatus = 'soon';
        else dueStatus = 'future';
    }
    const displayText = text
        .replace(/!!/g,'').replace(/!(?!!)/g,'')
        .replace(_tagPattern, '')
        .replace(/📅\s*\d{4}-\d{2}-\d{2}/g,'')
        .replace(/\s{2,}/g,' ').trim();
    const result = { text, displayText, isUrgent, isImportant, dueDate, dueStatus };
    _analyzeCache.set(key, result);
    return result;
}

        }
        
        // 全部展开/收起按钮
        if (e.target.classList.contains('nd-expand-all-btn')) {
            const allC = [...document.querySelectorAll('.nd-file-content[data-open]')];
            const allH = dash.querySelectorAll('.nd-fh');
            const allA = dash.querySelectorAll('[id$="-a"]');
            const allOpen = allC.every(c => c.dataset.open === '1');
            const show = !allOpen;
            const self = e.target;
            self.disabled = true;

            allC.forEach((c, i) => {
                animateFold(c, show, { padding: true });
                c.dataset.open = show ? '1' : '0';
            });

            allA.forEach(a => { a.style.transform = show ? 'rotate(90deg)' : ''; });
            allH.forEach(h => {
                h.style.borderBottom = show ? '1px solid var(--background-modifier-border)' : '1px solid transparent';
                h.dataset.open = show ? '1' : '0';
            });
            self.textContent = show ? '全部收起' : '全部展开';
            setTimeout(() => { self.disabled = false; }, 600);
            return;
        }
        
        // ⬜ 标记完成
        const cb = e.target.closest('.nd-cb');
        if (cb) {
            e.stopPropagation();
            const row = cb.closest('.nd-tr');
            const path = cb.dataset.path;
            const line = parseInt(cb.dataset.line);
            if (!row || !path || !line) return;
            
            cb.textContent = '✅';
            cb.style.transform = 'scale(1.5)';
            cb.style.color = '#22c55e';

            setTimeout(() => {

                row.style.transform = 'translateX(100%)';
                row.style.opacity = '0';
            }, 250);
            
            setTimeout(async () => {
                try {
                    const file = app.vault.getAbstractFileByPath(path);
                    if (file) {
                        let content = await app.vault.read(file);
                        content = content.replace(/\r\n/g, '\n');
                        const lines = content.split('\n');
                        const li = line - 1;
                        if (li >= 0 && li < lines.length) {
                            let l = lines[li];
                            l = l.replace(/^(\s*[-*+]\s*)\[\s\](\s*)/, '$1[x]$2');
                            lines[li] = l;

                    updateAllBtnText();
                } catch(err) {
                    console.error('标记失败:', err);
                    row.style.transform = '';
                    row.style.opacity = '1';
                    cb.textContent = '⬜';
                    cb.style.transform = '';
                    cb.style.color = '';
                }
            }, 650);
        }
    });
}

function isInViewport(el) {
    if (!el) return false;
    const rect = el.getBoundingClientRect();
    return rect.top < window.innerHeight && rect.bottom > 0;
}

// 首屏可见的容器立即加载（延迟一帧确保 DOM 已 paint）

    const heatEl = document.getElementById('nd-heat-container');
    const taskEl = document.getElementById('nd-task-container');
    if (heatEl && isInViewport(heatEl)) lazyLoadHeatmap();
    if (taskEl && isInViewport(taskEl)) lazyLoadTaskBoard();
    // 不在首屏的容器使用 IntersectionObserver 懒加载
    const lazyObserver = new IntersectionObserver((entries) => {
        entries.forEach(entry => {
            if (entry.isIntersecting) {
                const id = entry.target.id;
                if (id === 'nd-heat-container') lazyLoadHeatmap();
                else if (id === 'nd-task-container') lazyLoadTaskBoard();
                lazyObserver.unobserve(entry.target);
            }
        });
    }, { threshold: 0.1 });
    if (heatEl && !heatEl.dataset.loaded) lazyObserver.observe(heatEl);
    if (taskEl && !taskEl.dataset.loaded) lazyObserver.observe(taskEl);
});
