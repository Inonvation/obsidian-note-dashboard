// 🎯 核心策略：不在 preview 模式时，立即切换并 return，不做任何渲染。
// 切换触发的 Dataview 重执行会在 preview 模式下做一次干净渲染。

// ── 强制切换到 preview 模式（确保看板始终以阅读模式展示）──
// 增强版：过滤掉非标准 leaf（如 empty/ghost leaf），避免 setViewState 调用失败
function forcePreviewMode() {
    dv.app.workspace.iterateAllLeaves(l => {
        if (!l?.view?.file) return;
        if (l.view.file.path !== C.selfPath) return;
        // 跳过还没有完整 viewState 的 leaf
        const st = l.getViewState();
        if (!st || !st.state || typeof st.state.mode === 'undefined') return;
        if (st.state.mode !== 'preview') {
            l.setViewState({ ...st, state: { ...(st.state || {}), mode: 'preview' } });
        }
    });
}

// 检查当前 leaf 是否在 preview 模式
let ndIsPre = false;
dv.app.workspace.iterateAllLeaves(l => {
    if (l?.view?.file?.path !== C.selfPath) return;
    if (l.getViewState()?.state?.mode === 'preview') ndIsPre = true;
});

// 🎯 核心策略：不在 preview 模式时，跳过 DOM 写入并延后切换，保证重执行一次干净渲染
// 流程：首次打开 → 非preview → 清空容器 → 切换preview → 脚本结束
//       → 切换生效 → Obsidian 重载leaf → 脚本重执行 → 已在preview → 渲染
//
// 🔧 v3.1 修复：用 onLayoutReady + requestAnimationFrame + 重试兜底
// 解决浅色模式初始化过快导致 setTimeout(fn,0) 回调执行时 workspace 未就绪的问题
if (!ndIsPre) {
    dv.container.innerHTML = '';  // 清空容器，不让用户看到任何东西
    let retries = 0;
    const MAX_RETRIES = 5;
    const switchToPreview = () => {
        forcePreviewMode();
        // 检查是否真的生效了（通过比对当前 mode），未生效则重试
        let switched = false;
        dv.app.workspace.iterateAllLeaves(l => {
            if (l?.view?.file?.path !== C.selfPath) return;
            if (l.getViewState()?.state?.mode === 'preview') switched = true;
        });
        if (!switched && retries < MAX_RETRIES) {
            retries++;
            setTimeout(switchToPreview, 50 * retries);  // 退避重试 50ms, 100ms, 150ms...
        }
    };
    // 用 dv.app.workspace.onLayoutReady 确保 workspace 可安全操作
    dv.app.workspace.onLayoutReady(() => {
        requestAnimationFrame(() => {
            switchToPreview();
        });
    });
    return;
}

            requestAnimationFrame(() => forcePreviewMode());
        }
    });
}

// 统一折叠/展开动画（消除 4 处重复的 max-height + opacity 过渡逻辑）
// show: true=展开, false=折叠
// opts.childSelector: 展开时子元素交错入场的选择器
// opts.childDelay: 子元素交错间隔(ms)，默认 30
// opts.padding: 是否过渡 padding，默认 false
// opts.onDone: 动画完成回调
function animateFold(el, show, opts = {}) {
    if (el.classList.contains('nd-show') === show) return;
    const { childSelector, childDelay = 30, padding = false, onDone } = opts;
    const cleanProps = ['transition', 'maxHeight', 'opacity'].concat(padding ? ['paddingTop', 'paddingBottom'] : []);

    function cleanup() {
        cleanProps.forEach(p => el.style[p] = '');
        el.removeEventListener('transitionend', cleanup);
        if (onDone) onDone();
    }

    if (show) {
        el.classList.add('nd-show');
        el.style.maxHeight = '0px';
        el.style.opacity = '0';
        if (padding) { el.style.paddingTop = '0px'; el.style.paddingBottom = '0px'; }
        void el.offsetHeight;
        const h = el.scrollHeight;
        el.style.transition = 'max-height .35s cubic-bezier(.4,0,.2,1), opacity .2s ease'
            + (padding ? ', padding .3s ease' : '');
        el.style.maxHeight = h + 'px';
        el.style.opacity = '1';
        if (padding) { el.style.paddingTop = '4px'; el.style.paddingBottom = '6px'; }
        if (childSelector) {
            el.querySelectorAll(childSelector).forEach((row, i) => {
                row.style.opacity = '0';
                row.style.transform = 'translateY(-6px)';
                row.style.transition = 'none';
                setTimeout(() => {
                    row.style.transition = 'opacity .25s ease, transform .25s ease';
                    row.style.opacity = '1';
                    row.style.transform = '';
                }, 40 + i * childDelay);
            });
        }
        el.addEventListener('transitionend', cleanup, { once: true });
    } else {
        const h = el.scrollHeight;
        el.style.maxHeight = h + 'px';
        el.style.transition = 'none';
        if (padding) { el.style.paddingTop = padding ? '4px' : ''; el.style.paddingBottom = padding ? '6px' : ''; }
        void el.offsetHeight;
        el.style.transition = 'max-height .3s cubic-bezier(.4,0,.2,1), opacity .2s ease .1s'
            + (padding ? ', padding .25s ease .05s' : '');
        el.style.maxHeight = '0px';
        el.style.opacity = '0';
        if (padding) { el.style.paddingTop = '0px'; el.style.paddingBottom = '0px'; }
        el.addEventListener('transitionend', () => {
            el.classList.remove('nd-show');
            cleanup();
        }, { once: true });
    }
}

function bindChipInputs(panel) {
    panel.querySelectorAll('[data-nd-chip]').forEach(container => {
        const name = container.dataset.ndChip;
        const input = container.querySelector(`[data-nd-chip-input="${name}"]`);
        input.addEventListener('keydown', e => {
            if (e.key !== 'Enter' && e.key !== ',') return;
            e.preventDefault();
            const val = input.value.trim().replace(/,$/, '');
            if (!val) return;
            const chips = getChipValues(panel, name);
            if (chips.includes(val)) { input.value = ''; return; }
            chips.push(val);
            renderChips(container, name, chips);
            input.value = '';
        });
        container.addEventListener('click', e => {
            const del = e.target.closest('[data-nd-chip-del]');
            if (del) {
                const idx = parseInt(del.dataset.ndChipDel);
                const chips = getChipValues(panel, name);
                chips.splice(idx, 1);
                renderChips(container, name, chips);
                return;
            }
            input.focus();
        });
    });
}

function getChipValues(panel, name) {
    const container = panel.querySelector(`[data-nd-chip="${name}"]`);
    return [...container.querySelectorAll('[data-nd-chip-val] span:first-child')].map(s => s.textContent);
}

function renderChips(container, name, values) {
    const input = container.querySelector(`[data-nd-chip-input="${name}"]`);
    container.querySelectorAll('[data-nd-chip-val]').forEach(el => el.remove());
    values.forEach((v, i) => {
        const chip = document.createElement('span');
        chip.dataset.ndChipVal = i;
        chip.style.cssText = 'display:inline-flex;align-items:center;gap:3px;padding:2px 8px;border-radius:6px;background:var(--interactive-accent);color:#fff;font-size:11px;font-weight:500;';
        chip.innerHTML = `<span>${esc(v)}</span><span data-nd-chip-del="${i}" style="cursor:pointer;font-size:13px;line-height:1;opacity:.7;">×</span>`;
        container.insertBefore(chip, input);
    });
}

function collectFormValues(panel) {
    const newCfg = { ...DEFAULT_CONFIG };
    panel.querySelectorAll('[data-nd-field]').forEach(input => {
        const name = input.dataset.ndField;
        if (input.type === 'number') {
            const val = parseFloat(input.value);
            if (!isNaN(val)) {
                if (name.includes('[')) {
                    const [arr, idx] = name.replace(']', '').split('[');
                    if (!Array.isArray(newCfg[arr])) newCfg[arr] = [...DEFAULT_CONFIG[arr]];
                    newCfg[arr][parseInt(idx)] = val;
                } else {
                    newCfg[name] = val;
                }
            }
        } else {
            newCfg[name] = input.value.trim();
        }
    });
    newCfg.exclude = getChipValues(panel, 'exclude');
    if (newCfg.exclude.length === 0) newCfg.exclude = [...DEFAULT_CONFIG.exclude];
    newCfg.taskTags = getChipValues(panel, 'taskTags');
    const checked = panel.querySelector('input[name="nd-color"]:checked');
    if (checked) newCfg.colorScheme = checked.value;
    newCfg.heatLevels.sort((a, b) => a - b);
    return newCfg;
}

const allPages = dv.pages('').where(p =>
    !C.exclude.some(f => p.file.folder?.includes(f)) && p.file.path !== C.selfPath
);

// ── 内容缓存：加载真实内容，按 Obsidian 词数算法统计（非 file.size 估算）──
// 算法说明：中文每字1词，英文按空格分词
const WordCache = {
    _cache: null, _key: '',
    _keyOf(pages) { return pages.values.map(p => p.file.path + ':' + p.file.size + ':' + (p.file.mtime||0)).join('|'); },
    // 计算纯正文词数（与 Obsidian 状态栏算法一致）
    _wordCount(raw) {
        if (!raw) return 0;
        // 剥掉 YAML、代码块、标记语法，只留正文
        let t = raw.replace(/^---[\s\S]*?---\n*/,'').replace(/```[\s\S]*?```/g,'').replace(/`[^`]+`/g,'')
            // ![[嵌入文件]] → 完全移除（不计词）
            .replace(/!\[\[[^\]]*\]\]/g,'')
            // [[笔记]] → 笔记名；[[笔记#标题]] → 笔记名（剥掉锚点）；[[笔记|别名]] → 笔记名
            .replace(/\[\[([^\]|#^]+)(?:[#|][^\]]*)?\]\]/g,'$1')
            // [文字](链接) → 只保留文字
            .replace(/\[([^\]]*)\]\([^)]+\)/g,'$1')
            .replace(/(\*{1,3}|_{1,3})(.*?)\1/g,'$2').replace(/^#+\s*/gm,'')
            .replace(/^[\s]*[-*+]\s+/gm,'').replace(/^[\s]*\d+\.\s+/gm,'').replace(/^(>)\s*/gm,'');
        // 中日韩字符：每个算 1 词
        const cjk = (t.match(/[\u4e00-\u9fff\u3400-\u4dbf\uf900-\ufaff]/g) || []).length;
        // 非中日韩部分：按空白分词
        const other = t.replace(/[\u4e00-\u9fff\u3400-\u4dbf\uf900-\ufaff]/g,' ').split(/\s+/).filter(w=>w).length;
        return cjk + other;
    },
    // 完整加载：逐文件计算精确词数
    async load(pages) {
        const k = this._keyOf(pages);
        if (this._key === k && this._cache) return this._cache;
        // ⚡ 性能防护：如果笔记数量 > 200，改用 file.size 估算（避免大量 IO 阻塞）
        const tooMany = pages.length > cfg.estThreshold;
        if (tooMany) {
            const wordByPath = new Map();
            let totalWords = 0;
            for (const p of pages.values) {
                const est = Math.round((p.file.size || 0) / cfg.estCoeff);
                wordByPath.set(p.file.path, est);
                totalWords += est;
            }
            this._key = k; this._cache = { wordByPath, totalWords };
            return this._cache;
        }
        const raw = await Promise.all(pages.values.map(async p => {
            try { return await dv.io.load(p.file.path); } catch { return ''; }
        }));
        const wordByPath = new Map();
        let totalWords = 0;
        for (let i = 0; i < pages.length; i++) {
            const wc = this._wordCount(raw[i] || '');
            wordByPath.set(pages.values[i].file.path, wc);
            totalWords += wc;
        }
        this._key = k; this._cache = { wordByPath, totalWords };
        return this._cache;
    }
};

// 计算合适的Y轴刻度（取整数值）
function niceScale(maxVal, ticks) {
    if (maxVal <= 0) return { max: 100, step: 25, ticks: [0, 25, 50, 75, 100] };
    const rawStep = maxVal / ticks;
    const pow10 = Math.pow(10, Math.floor(Math.log10(rawStep)));
    const candidates = [pow10, pow10 * 2, pow10 * 5, pow10 * 10];
    let niceStep = candidates[0];
    for (const c of candidates) { if (c * ticks >= maxVal) { niceStep = c; break; } }
    const tickValues = [];
    for (let i = 0; i <= ticks; i++) tickValues.push(Math.round(i * niceStep));
    return { max: niceStep * ticks, step: niceStep, ticks: tickValues };
}

// 通用toggle切换函数（消除重复的滑动指示器+颜色切换逻辑）
function setupToggle(toggleEl, onSwitch) {
    const indicator = toggleEl.querySelector('[data-toggle-indicator]');
    const spans = toggleEl.querySelectorAll('span[data-toggle-view]');
    let currentView = null;
    spans.forEach(sp => {
        sp.addEventListener('click', function(e) {
            e.stopPropagation();
            const view = this.dataset.toggleView;
            if (view === currentView) return;
            currentView = view;
            if (indicator) indicator.style.left = view === spans[spans.length-1].dataset.toggleView ? 'calc(50%)' : '2px';
            spans.forEach(s => { s.style.color = s === this ? '#fff' : 'var(--text-muted)'; });
            onSwitch(view, this);
        });
    });
    // 初始化当前视图
    currentView = spans[0]?.dataset.toggleView;
}

        const pct = total > 0 ? d.words / total : 0;
        const angle = Math.max(pct * 360, 2);
        const startDeg = currentDeg, endDeg = currentDeg + angle;
        const sRad = startDeg * Math.PI / 180, eRad = endDeg * Math.PI / 180;
        const path = `M${cx},${cy} L${cx+r*Math.cos(sRad)},${cy+r*Math.sin(sRad)} A${r},${r} 0 ${angle>180?1:0},1 ${cx+r*Math.cos(eRad)},${cy+r*Math.sin(eRad)} Z`;
        const midDeg = (startDeg + endDeg) / 2, mRad = midDeg * Math.PI / 180;
        const lx = cx + r * Math.cos(mRad), ly = cy + r * Math.sin(mRad);
        const lx2 = cx + (r + lp) * Math.cos(mRad), ly2 = cy + (r + lp) * Math.sin(mRad);
        slices.push({ path, color: C.colors[i % C.colors.length], pct, angle, lx, ly, lx2, ly2, name: d.name });
        currentDeg += angle;
    });
    // ── 标签防碰撞 ──
    const labels = slices.map(s => ({ ...s, ty: s.ly2 }));
    labels.sort((a, b) => a.ty - b.ty);
    const minGap = 28;
    for (let i = 1; i < labels.length; i++) {
        if (labels[i].ty - labels[i - 1].ty < minGap) labels[i].ty = labels[i - 1].ty + minGap;
    }
    const svgH = 220;
    if (labels.length) {
        const over = (labels[labels.length - 1].ty + 10) - (svgH - 4);
        if (over > 0) labels.forEach(l => l.ty -= over);
    }
    const labelMap = {};
    labels.forEach(l => { labelMap[l.name] = l.ty; });

    const uid = 'fp' + Date.now();
    let svg = `<svg viewBox="0 0 360 ${svgH}" style="width:100%;height:auto;display:block;max-height:220px;">`;
    svg += `<defs><filter id="${uid}sh" x="-15%" y="-15%" width="130%" height="130%">`;
    svg += `<feDropShadow dx="0" dy="1" stdDeviation="3" flood-color="rgba(0,0,0,.1)"/></filter></defs>`;
    // 饼图
    svg += `<g filter="url(#${uid}sh)">`;
    slices.forEach((s, i) => svg += `<path data-nd-pie="${i}" d="${s.path}" fill="${s.color}" stroke="var(--background-primary)" stroke-width="2"/>`);
    svg += `</g>`;
    svg += `<circle cx="${cx}" cy="${cy}" r="2.5" fill="var(--background-primary)" opacity="0.45"/>`;
    // 直引线 + 标签
    slices.forEach(s => {
        const ty = labelMap[s.name];
        const right = s.lx2 >= cx;
        const tx = right ? s.lx2 + 5 : s.lx2 - 5;
        const anchor = right ? 'start' : 'end';
        svg += `<line x1="${s.lx.toFixed(1)}" y1="${s.ly.toFixed(1)}" x2="${tx.toFixed(1)}" y2="${ty.toFixed(1)}" stroke="${s.color}" stroke-width="0.8" opacity="0.4"/>`;
        svg += `<circle cx="${s.lx.toFixed(1)}" cy="${s.ly.toFixed(1)}" r="2" fill="${s.color}" stroke="var(--background-primary)" stroke-width="1"/>`;
        svg += `<text x="${tx.toFixed(1)}" y="${ty.toFixed(1)}" text-anchor="${anchor}" dominant-baseline="middle" font-size="11" fill="var(--text-normal)" font-weight="600">${s.name}</text>`;
        svg += `<text x="${tx.toFixed(1)}" y="${(ty + 12).toFixed(1)}" text-anchor="${anchor}" dominant-baseline="middle" font-size="8.5" fill="var(--text-muted)">${(s.pct * 100).toFixed(1)}%</text>`;
    });
    svg += `</svg>`;
    return { svg, legend: '', uid };
}

// ── 辅助函数 ──
function analyzeTask(task) {
    const text = task.text || '';
    const isUrgent = /!!/.test(text);
    const isImportant = (/!(?!!)/.test(text) && !/!!/.test(text)) || /#important/i.test(text);
    const dueMatch = text.match(/📅\s*(\d{4}-\d{2}-\d{2})/);
    const dueDate = dueMatch ? new Date(dueMatch[1]) : null;
    let dueStatus = 'none';
    if (dueDate) {
        const diff = Math.ceil((dueDate - C.now) / (1000*60*60*24));
        if (diff < 0) dueStatus = 'overdue';
        else if (diff === 0) dueStatus = 'today';
        else if (diff <= 2) dueStatus = 'soon';
        else dueStatus = 'future';
    }
    const tagPattern = cfg.taskTags.map(t => t.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')).join('|');
    const displayText = text
        .replace(/!!/g,'').replace(/!(?!!)/g,'')
        .replace(new RegExp(tagPattern, 'gi'), '')
        .replace(/📅\s*\d{4}-\d{2}-\d{2}/g,'')
        .replace(/\s{2,}/g,' ').trim();
    return { text, displayText, isUrgent, isImportant, dueDate, dueStatus };
}

function badge(text, color, bg, pulse) {
    const a = pulse ? 'animation:ndPulseTag 2s ease-in-out infinite;' : '';
    return `<span style="display:inline-flex;font-size:9px;padding:0 6px;border-radius:6px;background:${bg};color:${color};font-weight:600;line-height:1.8;${a}">${text}</span>`;
}
function dueBadge(t) {
    const a = analyzeTask(t); if (!a.dueDate) return '';
    const m = {overdue:{l:'逾期',c:'#ef4444',b:'rgba(239,68,68,0.15)'},today:{l:'今日到期',c:'#f97316',b:'rgba(249,115,22,0.15)'},soon:{l:'即将到期',c:'#ca8a04',b:'rgba(234,179,8,0.12)'},future:{l:`${a.dueDate.getMonth()+1}/${a.dueDate.getDate()}`,c:'var(--text-faint)',b:'transparent'}}[a.dueStatus];
    return m ? badge(`📅 ${m.l}`, m.c, m.b, a.dueStatus==='overdue'||a.dueStatus==='today') : '';
}
function priBadge(t) {
    const a = analyzeTask(t);
    if (a.isUrgent) return badge('🔴 紧急','#ef4444','rgba(239,68,68,0.15)',true);
    if (a.isImportant) return badge('🟠 重要','#f97316','rgba(249,115,22,0.12)',false);
    return '';
}

// ═══ 事件委托（用 data-nd-eaid 动态读取当前按钮 id，避免重执行后旧监听器匹配失效） ═══
// 关键：每次执行都更新 dash.dataset.ndEaid = 当前 eaId，旧监听器通过 this.dataset.ndEaid 读取最新值
function bindTaskBoardEvents() {
    const dash = document.getElementById('nd-dash');
    if (!dash) return;
    // 更新当前按钮 id 到 dash 上（每次重执行都刷新，旧监听器通过 this.dataset 读到最新值）
    dash.dataset.ndEaid = eaId;
    // 仅首次挂监听器
    if (dash.dataset.ndBound === '1') return;
    dash.dataset.ndBound = '1';

    // 唯一的事件委托监听器
    dash.addEventListener('click', function(e) {
        // ── 文件头点击展开/折叠 ──
        const fh = e.target.closest('.nd-fh');
        if (fh) {
            const tid = fh.dataset.target;
            const c = document.getElementById(tid);
            const a = document.getElementById(tid + '-a');
            if (!c) return;
            const isOpen = c.dataset.open === '1';
            
            toggleFileContent(c, !isOpen);
            
            if (a) a.style.transform = isOpen ? '' : 'rotate(90deg)';
            fh.style.borderBottom = isOpen ? '1px solid transparent' : '1px solid var(--background-modifier-border)';
            c.dataset.open = isOpen ? '0' : '1';
            fh.dataset.open = isOpen ? '0' : '1';
            updateAllBtnText();
            return;
        }
        
        // ── 全部展开/收起按钮 ──
        if (e.target.id === eaId) {
            const allC = [...document.querySelectorAll('.nd-file-content[data-open]')];
            const allH = dash.querySelectorAll('.nd-fh');
            const allA = dash.querySelectorAll('[id$="-a"]');
            const allOpen = allC.every(c => c.dataset.open === '1');
            const show = !allOpen;
            const self = e.target;
            self.disabled = true;

            allC.forEach((c, i) => {
                animateFold(c, show, { padding: true });
                c.dataset.open = show ? '1' : '0';
            });

            allA.forEach(a => { a.style.transform = show ? 'rotate(90deg)' : ''; });
            allH.forEach(h => {
                h.style.borderBottom = show ? '1px solid var(--background-modifier-border)' : '1px solid transparent';
                h.dataset.open = show ? '1' : '0';
            });
            self.textContent = show ? '全部收起' : '全部展开';
            setTimeout(() => { self.disabled = false; }, 600);
            return;
        }
        
        // ── ⬜ 标记完成 ──
        const cb = e.target.closest('.nd-cb');
        if (cb) {
            e.stopPropagation();
            const row = cb.closest('.nd-tr');
            const path = cb.dataset.path;
            const line = parseInt(cb.dataset.line);
            if (!row || !path || !line) return;
            
            cb.textContent = '✅';
            cb.style.transform = 'scale(1.5)';
            cb.style.color = '#22c55e';
            cb.style.transition = 'transform .2s cubic-bezier(.4,0,.2,1)';
            
            setTimeout(() => {
                cb.style.transform = 'scale(1)';
                row.style.transition = 'none';
                row.style.transform = 'translateX(0)';
                row.style.opacity = '1';
                void row.offsetWidth;
                row.style.transition = 'transform .35s cubic-bezier(.4,0,.2,1), opacity .35s ease';
                row.style.transform = 'translateX(100%)';
                row.style.opacity = '0';
            }, 250);
            
            setTimeout(async () => {
                try {
                    const file = app.vault.getAbstractFileByPath(path);
                    if (file) {
                        let content = await app.vault.read(file);
                        content = content.replace(/\r\n/g, '\n');
                        const lines = content.split('\n');
                        const li = line - 1;
                        if (li >= 0 && li < lines.length) {
                            let l = lines[li];
                            l = l.replace(/^(\s*[-*+]\s*)\[\s\](\s*)/, '$1[x]$2');
                            lines[li] = l;
                            await app.vault.modify(file, lines.join('\n'));
                        }
                    }
                    row.remove();
                    updateAllBtnText();
                } catch(err) {
                    console.error('标记失败:', err);
                    row.style.transform = '';
                    row.style.opacity = '1';
                    cb.textContent = '⬜';
                    cb.style.transform = '';
                    cb.style.color = '';
                }
            }, 650);
        }
    });
}
